def analyze_fuel_type(df):
    """
    'fuel_type' kolonunu analiz eder, sınıflandırır, temizler ve makine öğrenmesi modellerine uygun hale getirir.
    
    İşlevleri:
    - 'fuel_type' dağılımını gösterir.
    - Eksik değerleri doldurur ve kategorileri birleştirir.
    - 'Other' kategorisini siler, 'Unknown' değerlerini 'Electric' olarak dönüştürür.
    - Yeni 'fuel_group' kolonu oluşturur.
    - 'fuel_group' için One-Hot Encoding (dummy değişkenler) uygular.
    """

    print("\n--- Adım: Fuel Type Analizi ve Encoding ---")

    # 1️⃣ Yakıt türü dağılımı
    print("\nYakıt Türü Dağılımı:")
    print(df['fuel_type'].value_counts(dropna=False))
    print(f"\nEksik değer sayısı: {df['fuel_type'].isna().sum()}")

    # 2️⃣ Eksik değerleri doldur
    df['fuel_type'] = df['fuel_type'].fillna('Unknown')

    # 3️⃣ Yakıt türlerini sınıflandır
    def classify_fuel_type(ftype):
        ftype = str(ftype).lower()

        if any(x in ftype for x in ['gas', 'petrol']):
            return 'Gasoline'
        elif 'diesel' in ftype:
            return 'Diesel'
        elif 'hybrid' in ftype and 'plug' not in ftype:
            return 'Hybrid'
        elif 'plug' in ftype or 'electric' in ftype:
            return 'Electric'
        elif 'flex' in ftype:
            return 'Flex Fuel'
        elif ftype == 'unknown':
            return 'Unknown'
        else:
            return 'Other'

    df['fuel_group'] = df['fuel_type'].apply(classify_fuel_type)

    # 4️⃣ Unknown → Electric, Other → sil
    before_len = len(df)
    df['fuel_group'] = df['fuel_group'].replace('Unknown', 'Electric')
    df = df[df['fuel_group'] != 'Other']
    after_len = len(df)
    print(f"\n{before_len - after_len} satır 'Other' kategorisinden silindi.")
    print("Tüm 'Unknown' değerler 'Electric' kategorisine eklendi.")

    # 5️⃣ One-Hot Encoding
    fuel_dummies = pd.get_dummies(df['fuel_group'], prefix='fuel', drop_first=False)
    df = pd.concat([df, fuel_dummies], axis=1)

    print("\nOluşturulan encoding sütunları:")
    print(list(fuel_dummies.columns))

    # 6️⃣ Kontrol için dağılımı yazdır
    print("\nYeni Gruplandırılmış Yakıt Türü Dağılımı (fuel_group):")
    print(df['fuel_group'].value_counts())

    return df
